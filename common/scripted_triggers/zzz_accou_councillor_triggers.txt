accou_can_be_marshal_trigger = {
	exists = $COURT_OWNER$
	can_be_marshal_trigger = { COURT_OWNER = $COURT_OWNER$ }
	tgp_is_ceremonial_liege_trigger = no
	#AGOT Disabled
	#NOT = { this = character:528 }
	#AGOT Added
	OR = {
		NOT = { $COURT_OWNER$ = title:k_the_wall.holder }
		AND = {
			$COURT_OWNER$ = title:k_the_wall.holder
			NOT = { $COURT_OWNER$ = { has_trait = nightswatch } }
			NOT = { has_trait = nightswatch }
		}
		AND = {
			$COURT_OWNER$ = title:k_the_wall.holder
			title:k_the_wall.holder = {
				has_trait = nightswatch
				NOT = { has_character_flag = fr_assign_block }
			}
			has_trait = nightswatch
			has_trait = lifestyle_nw_ranger
			is_landed = no
			OR = {
				$COURT_OWNER$ = {
					NOT = {
						any_vassal = {
							has_character_flag = nw_ranger_title
							is_alive = yes
						}
					}
				}
				has_character_flag = nw_ranger_title
			}
		}
	}
}

accou_can_be_spymaster_trigger = {
	# valid_character
	exists = $COURT_OWNER$
	can_be_spymaster_trigger = { COURT_OWNER = $COURT_OWNER$ }
	trigger_if = {
		limit = {
			$COURT_OWNER$ = {
				is_landed = yes
				government_has_flag = government_has_merit
			}
		}
		trigger_if = {
			limit = {
				$COURT_OWNER$ = {
					OR = {
						has_realm_law = male_only_law
						has_realm_law = male_preference_law
					}
				}
			}
			is_male = yes
		}
		trigger_if = {
			limit = {
				$COURT_OWNER$ = {
					OR = {
						has_realm_law = female_only_law
						has_realm_law = female_preference_law
					}
				}
			}
			is_female = yes
		}
	}
	tgp_is_ceremonial_liege_trigger = no
	#AGOT Added
	NOT = { has_character_flag = nw_ranger_title }
}
